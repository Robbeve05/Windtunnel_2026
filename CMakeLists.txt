cmake_minimum_required(VERSION 3.16)
project(windtunnel_core C)
set(CMAKE_C_STANDARD 11)
add_library(windtunnel_core arduino/src/windtunnel.c arduino/src/protocol.c)
target_include_directories(windtunnel_core PUBLIC arduino/include)
if(NOT MSVC)
  target_compile_options(windtunnel_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(windtunnel_core PUBLIC m)
endif()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE windtunnel_core)
if(MSVC)
  target_compile_options(test_core PRIVATE /UNDEBUG)
else()
  target_compile_options(test_core PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME core COMMAND test_core)
